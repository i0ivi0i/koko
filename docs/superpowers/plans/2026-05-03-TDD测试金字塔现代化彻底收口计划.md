const 完成态顶层文件硬上限 = 1000;

function 收集完成态顶层超限(relativePath, lines) {
  if (!是Rust顶层集成测试(relativePath) && !relativePath.startsWith("frontend/tests/")) {
    return [];
  }
  if (lines.length <= 完成态顶层文件硬上限) {
    return [];
  }
  return [{
    file: relativePath,
    label: "不满足 spec 完成态",
    detail: `${relativePath}: ${lines.length} lines > ${完成态顶层文件硬上限}`,
  }];
}

function 收集真实浏览器层缺失() {
  const hasPackage = existsSync(join(仓库根目录, "e2e/package.json"));
  const testDir = join(仓库根目录, "e2e/tests");
  const specs = existsSync(testDir)
    ? 枚举文件(testDir, path => path.endsWith(".spec.ts"))
    : [];
  if (hasPackage && specs.length >= 3) return [];
  return [{
    file: "e2e/",
    label: "缺少 repo-native Playwright 烟测层",
    detail: "e2e/package.json 或最小 smoke specs 不完整",
  }];
}

const violations = [
  ...records.flatMap(record => [
    ...收集行数违规(record.relativePath, record.lineArray),
    ...收集完成态顶层超限(record.relativePath, record.lineArray),
    ...收集only违规(record.relativePath, record.lineArray),
    ...收集skip违规(record.relativePath, record.lineArray),
  ]),
  ...收集真实浏览器层缺失(),
];

- [ ] **Step 6: 用项目批准链路包装 e2e 执行**

```powershell
function Stop-ProcTree([int]$RootProcessId) {
  $children = Get-CimInstance Win32_Process | Where-Object { $_.ParentProcessId -eq $RootProcessId }
  foreach ($child in $children) {
    Stop-ProcTree -RootProcessId ([int]$child.ProcessId)
  }
  Stop-Process -Id $RootProcessId -Force -ErrorAction SilentlyContinue
}

$launcher = Start-Process powershell.exe -ArgumentList @(
  "-NoProfile","-ExecutionPolicy","Bypass","-File",".\\run.ps1",
  "-DisableLocalHttpsBootstrap","-DisableAutoOptimizeCleanup"
) -PassThru -WindowStyle Hidden

try {
  for ($i = 0; $i -lt 90; $i++) {
    if (Get-NetTCPConnection -LocalPort 8080 -State Listen -ErrorAction SilentlyContinue) {
      break
    }
    Start-Sleep -Seconds 1
  }
  pnpm --dir e2e exec playwright install chromium
  $env:KOKO_E2E_BASE_URL = "http://127.0.0.1:8080"
  pnpm --dir e2e test
}
finally {
  Stop-ProcTree -RootProcessId ([int]$launcher.Id)
}
```
