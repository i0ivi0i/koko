## 关键公式速查

```
cols       = 2
cellWidth  = (maxWidth - spacing) / cols
cellHeight = cellWidth * 3 / 2                    // 2:3
rows       = Math.ceil(count / cols)
totalHeight = rows * cellHeight + (rows - 1) * spacing

正常卡片:
  x = col * (cellWidth + spacing)
  y = row * (cellHeight + spacing)
  width = cellWidth, height = cellHeight

尾行单张 (count 为奇数):
  x = 0, y = lastRow * (cellHeight + spacing)
  width = maxWidth, height = cellHeight
```

默认配置：`maxWidth = 384, spacing = 8`
→ `cellWidth = 188, cellHeight = 282`

```typescript
function 断言无溢出(result: 拼贴结果, maxWidth: number): void {
  for (const item of result.items) {

function 断言卡片比例为2比3(result: 拼贴结果, count: number): void {
  const cols = 2;
  const lastRowCols = count % cols === 0 ? cols : count % cols;
  const lastRowIndex = count - lastRowCols;

  for (let i = 0; i < result.items.length; i++) {
    const item = result.items[i]!;
    if (lastRowCols === 1 && i === lastRowIndex) {
      // 尾行单张拉满，比例不是 2:3，跳过
      continue;

    const ratio = item.width / item.height;
    // 2:3 = 0.6667，容忍取整 ±0.01
    expect(ratio).toBeGreaterThan(0.65);
    expect(ratio).toBeLessThan(0.69);
  }
}
```

```typescript
/* ── 统一网格布局（≥2 张） ── */

/**
 * 统一 2 列、2:3 卡片比例的均匀网格。
 * 奇数尾行单张拉满宽度，行高与正常行一致。
 *
 * 不依赖媒体原始尺寸 —— 卡片形状由网格决定，
 * 视频/图片内容通过 object-fit: cover 裁切适配。
 */
function 统一网格布局(
  count: number,
  maxWidth: number,
  spacing: number
): 拼贴结果 {
  const cols = 2;
  const cellWidth = Math.round((maxWidth - spacing) / cols);
  const cellHeight = Math.round((cellWidth * 3) / 2);
  const rows = Math.ceil(count / cols);
  const lastRowCols = count - (rows - 1) * cols;

  const geometries: Array<{ x: number; y: number; width: number; height: number }> = [];

  for (let i = 0; i < count; i++) {
    const row = Math.floor(i / cols);
    const col = i % cols;
    const isLastRow = row === rows - 1;
    const isLastRowSingle = isLastRow && lastRowCols === 1;

    geometries.push({
      x: isLastRowSingle ? 0 : col * (cellWidth + spacing),
      y: row * (cellHeight + spacing),
      width: isLastRowSingle ? maxWidth : cellWidth,
      height: cellHeight,
    });
  }

  return 转换为拼贴结果(geometries, maxWidth);
}
```
